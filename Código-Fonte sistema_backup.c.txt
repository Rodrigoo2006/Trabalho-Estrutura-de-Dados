#include <stdio.h>
#include <stdlib.h>
#include <string.h>

#define MAX_NOME 64
#define MAX_ALTERACAO 120
#define MAX_DESC 220

/* =========================================================
 * 1. PILHA (STACK - LIFO): Historico de Versoes de Backup
 * ========================================================= */
typedef struct NoBackup {
    int idBackup;
    char descricao[MAX_DESC];
    struct NoBackup* proximo;
} NoBackup;

NoBackup* topo = NULL;
int contadorBackups = 0;

/* =========================================================
 * 2. LISTA ENCADEADA: Catalogo Geral de Arquivos
 * ========================================================= */
typedef struct NoArquivo {
    char nome[MAX_NOME];
    char ultimaAlteracao[MAX_ALTERACAO];
    int revisoes;
    struct NoArquivo* proximo;
} NoArquivo;

NoArquivo* listaArquivos = NULL;

/* =========================================================
 * 3. FILA (QUEUE - FIFO): Processamento em Lote de Backups
 * ========================================================= */
typedef struct NoFila {
    char descricao[MAX_DESC];
    struct NoFila* proximo;
} NoFila;

typedef struct {
    NoFila* inicio;
    NoFila* fim;
} FilaBackup;

FilaBackup filaProcessamento = {NULL, NULL};

/* ================= FUNCOES AUXILIARES ================= */
void lerString(const char* prompt, char* destino, size_t tamanho) {
    printf("%s", prompt);
    if (fgets(destino, (int)tamanho, stdin) != NULL) {
        size_t len = strlen(destino);
        if (len > 0 && destino[len - 1] == '\n') {
            destino[len - 1] = '\0';
        } else {
            int ch;
            while ((ch = getchar()) != '\n' && ch != EOF);
        }
    } else {
        destino[0] = '\0';
    }
}

int lerInteiro(const char* prompt) {
    char buffer[32];
    int valor = -1;
    printf("%s", prompt);
    if (fgets(buffer, sizeof(buffer), stdin) != NULL) {
        if (sscanf(buffer, "%d", &valor) != 1) {
            return -1;
        }
    }
    return valor;
}

/* ================= OPERACOES DA FILA (FIFO) ================= */
void enfileirarBackup(const char* desc) {
    NoFila* novo = (NoFila*) malloc(sizeof(NoFila));
    if (novo == NULL) {
        printf("[ERRO] Falha de alocacao de memoria na fila!\n");
        return;
    }

    strncpy(novo->descricao, desc, MAX_DESC - 1);
    novo->descricao[MAX_DESC - 1] = '\0';
    novo->proximo = NULL;

    if (filaProcessamento.fim == NULL) {
        filaProcessamento.inicio = novo;
        filaProcessamento.fim = novo;
    } else {
        filaProcessamento.fim->proximo = novo;
        filaProcessamento.fim = novo;
    }
}

/* ================= OPERACOES DA PILHA (LIFO) ================= */
void criarBackup(const char* desc) {
    NoBackup* novo = (NoBackup*) malloc(sizeof(NoBackup));
    if (novo == NULL) {
        printf("[ERRO] Erro de alocacao de memoria!\n");
        return;
    }

    contadorBackups++;
    novo->idBackup = contadorBackups;
    strncpy(novo->descricao, desc, MAX_DESC - 1);
    novo->descricao[MAX_DESC - 1] = '\0';

    novo->proximo = topo;
    topo = novo;

    printf("[SUCESSO] Backup #%d criado com sucesso!\n", contadorBackups);
}

void consultarHistorico(void) {
    if (topo == NULL) {
        printf("[AVISO] Historico vazio.\n");
        return;
    }

    NoBackup* atual = topo;
    printf("\n===== HISTORICO DE VERSOES (PILHA LIFO) =====\n");
    while (atual != NULL) {
        printf("-> Backup ID: %d | Descricao: %s\n", atual->idBackup, atual->descricao);
        atual = atual->proximo;
    }
    printf("=============================================\n");
}

void restaurarBackup(void) {
    if (topo == NULL) {
        printf("[AVISO] Nenhum backup disponivel para restauracao.\n");
        return;
    }

    NoBackup* temp = topo;
    printf("[RESTAURACAO] Restaurando Backup ID %d: %s\n", temp->idBackup, temp->descricao);
    topo = topo->proximo;
    free(temp);
    printf("[SUCESSO] Sistema revertido para o estado anterior.\n");
}

/* ================= OPERACOES DA LISTA ENCADEADA ================= */
void cadastrarOuModificarArquivo(void) {
    char nome[MAX_NOME];
    char alteracao[MAX_ALTERACAO];
    char descFila[MAX_DESC];

    lerString("Digite o nome do arquivo (ex: main.c): ", nome, sizeof(nome));
    if (nome[0] == '\0') {
        printf("[AVISO] Nome de arquivo invalido.\n");
        return;
    }

    lerString("Digite a descricao da criacao/modificacao: ", alteracao, sizeof(alteracao));
    if (alteracao[0] == '\0') {
        strncpy(alteracao, "Alteracao sem descricao", sizeof(alteracao) - 1);
        alteracao[sizeof(alteracao) - 1] = '\0';
    }

    NoArquivo* atual = listaArquivos;
    while (atual != NULL) {
        if (strcmp(atual->nome, nome) == 0) {
            strncpy(atual->ultimaAlteracao, alteracao, MAX_ALTERACAO - 1);
            atual->ultimaAlteracao[MAX_ALTERACAO - 1] = '\0';
            atual->revisoes++;

            snprintf(descFila, sizeof(descFila), "[%s - rev.%d] %s",
                     atual->nome, atual->revisoes, atual->ultimaAlteracao);
            enfileirarBackup(descFila);

            printf("[SUCESSO] Arquivo '%s' atualizado (revisao %d) e adicionado a fila de backup!\n",
                   atual->nome, atual->revisoes);
            return;
        }
        atual = atual->proximo;
    }

    NoArquivo* novo = (NoArquivo*) malloc(sizeof(NoArquivo));
    if (novo == NULL) {
        printf("[ERRO] Falha de alocacao de memoria no catalogo!\n");
        return;
    }

    strncpy(novo->nome, nome, MAX_NOME - 1);
    novo->nome[MAX_NOME - 1] = '\0';
    strncpy(novo->ultimaAlteracao, alteracao, MAX_ALTERACAO - 1);
    novo->ultimaAlteracao[MAX_ALTERACAO - 1] = '\0';
    novo->revisoes = 1;

    novo->proximo = listaArquivos;
    listaArquivos = novo;

    snprintf(descFila, sizeof(descFila), "[%s - rev.1] %s",
             novo->nome, novo->ultimaAlteracao);
    enfileirarBackup(descFila);

    printf("[SUCESSO] Arquivo '%s' cadastrado no catalogo e adicionado a fila de backup!\n",
           novo->nome);
}

void listarArquivos(void) {
    if (listaArquivos == NULL) {
        printf("[AVISO] Nenhum arquivo cadastrado no catalogo.\n");
        return;
    }

    NoArquivo* atual = listaArquivos;
    printf("\n===== CATALOGO DE ARQUIVOS (LISTA ENCADEADA) =====\n");
    while (atual != NULL) {
        printf("- Arquivo: %s | Revisoes: %d | Ultima alteracao: %s\n",
               atual->nome, atual->revisoes, atual->ultimaAlteracao);
        atual = atual->proximo;
    }
    printf("==================================================\n");
}

void processarFilaBackups(void) {
    if (filaProcessamento.inicio == NULL) {
        printf("[AVISO] Nenhum backup pendente na fila de processamento.\n");
        return;
    }

    printf("\n===== PROCESSANDO FILA DE BACKUPS (FIFO) =====\n");
    while (filaProcessamento.inicio != NULL) {
        NoFila* temp = filaProcessamento.inicio;
        printf("Processando: %s\n", temp->descricao);
        criarBackup(temp->descricao);

        filaProcessamento.inicio = temp->proximo;
        free(temp);
    }
    filaProcessamento.fim = NULL;
    printf("==============================================\n");
}

/* ================= LIBERACAO DE MEMORIA ================= */
void liberarMemoria(void) {
    while (topo != NULL) {
        NoBackup* temp = topo;
        topo = topo->proximo;
        free(temp);
    }

    while (listaArquivos != NULL) {
        NoArquivo* temp = listaArquivos;
        listaArquivos = listaArquivos->proximo;
        free(temp);
    }

    while (filaProcessamento.inicio != NULL) {
        NoFila* temp = filaProcessamento.inicio;
        filaProcessamento.inicio = temp->proximo;
        free(temp);
    }
    filaProcessamento.fim = NULL;
}

/* ================= FUNCAO PRINCIPAL ================= */
int main(void) {
    int opcao;
    char descricao[MAX_DESC];

    do {
        printf("\n===== SISTEMA DE BACKUP DE ARQUIVOS =====\n");
        printf("1 - Cadastrar / Modificar Arquivo\n");
        printf("2 - Criar Versao de Backup (Push)\n");
        printf("3 - Consultar Historico de Backups\n");
        printf("4 - Restaurar Ultimo Backup (Pop)\n");
        printf("5 - Listar Arquivos Cadastrados\n");
        printf("6 - Processar Fila de Backups (FIFO)\n");
        printf("0 - Sair\n");

        opcao = lerInteiro("Escolha uma opcao: ");

        switch (opcao) {
            case 1:
                cadastrarOuModificarArquivo();
                break;
            case 2:
                lerString("Digite a descricao: ", descricao, sizeof(descricao));
                if (descricao[0] == '\0') {
                    printf("[AVISO] A descricao nao pode ficar vazia.\n");
                } else {
                    criarBackup(descricao);
                }
                break;
            case 3:
                consultarHistorico();
                break;
            case 4:
                restaurarBackup();
                break;
            case 5:
                listarArquivos();
                break;
            case 6:
                processarFilaBackups();
                break;
            case 0:
                printf("Encerrando o sistema e liberando memoria...\n");
                break;
            default:
                printf("[AVISO] Opcao invalida. Tente novamente.\n");
                break;
        }
    } while (opcao != 0);

    liberarMemoria();
    return 0;
}
